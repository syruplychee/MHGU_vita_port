cmake_minimum_required(VERSION 3.24)
project(engine_binary_reference C)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
foreach(kind IN ITEMS material state queue)
    add_executable(${kind}_fixture ${kind}_fixture.c)
    target_include_directories(${kind}_fixture PRIVATE "${REPO}/src/platform" "${REPO}/tests")
    target_compile_features(${kind}_fixture PRIVATE c_std_11)
    if(MSVC)
        target_compile_definitions(${kind}_fixture PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
endforeach()
target_sources(material_fixture PRIVATE "${REPO}/src/platform/engine_material.c")
target_sources(state_fixture PRIVATE "${REPO}/src/platform/engine_state.c")
target_sources(queue_fixture PRIVATE "${REPO}/src/platform/engine_material.c"
    "${REPO}/src/platform/engine_state.c" "${REPO}/src/platform/engine_queue_native.c")
